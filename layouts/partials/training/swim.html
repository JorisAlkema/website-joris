{{- /* Swimming (25 m pool): strokes per length against pace per 100 m, one dot per session; the last three months in the accent colour. Shows whether faster swimming comes with fewer strokes (longer stroke) or more (higher turnover). Args: swim, nl. */ -}} {{- $s := .swim }}{{ $nl := .nl -}} {{- $rows := where $s.sessions "strokes" "!=" nil -}}

{{ if $nl -}} {{ $s.count }} trainingen in een 25 m-bad sinds {{ time.Format "January 2006" (time.AsTime $s.since) }}, samen {{ lang.FormatNumber 1 $s.total_km }} km. {{- else -}} {{ $s.count }} sessions in a 25 m pool since {{ time.Format "January 2006" (time.AsTime $s.since) }}, {{ lang.FormatNumber 1 $s.total_km }} km in total. {{- end }}
